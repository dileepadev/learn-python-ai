The rule that follows: **never put a live model call in a unit test.** It is slow, costs money,
and fails for reasons unrelated to your change.

## pytest

<Run id="prod-test-2" code={`from pathlib import Path
import pytest

Path("chunking.py").write_text('''
def chunk_words(text, size=10, overlap=2):
    if overlap >= size:
        raise ValueError("overlap must be smaller than size")
    words = text.split()
    if not words:
        return []
    step = size - overlap
    chunks = []
    for start in range(0, len(words), step):
        piece = words[start:start + size]
        if not piece:
            break
        chunks.append(" ".join(piece))
        if start + size >= len(words):
            break
    return chunks
''')

Path("test_chunking.py").write_text('''
import pytest
from chunking import chunk_words

def test_short_text_is_one_chunk():
    assert chunk_words("one two three", size=10) == ["one two three"]

def test_empty_text_gives_no_chunks():
    assert chunk_words("") == []

def test_chunks_overlap():
    chunks = chunk_words(" ".join(f"w{n}" for n in range(1, 21)), size=10, overlap=2)
    assert chunks[0].split()[-2:] == chunks[1].split()[:2]

def test_no_word_is_lost():
    text = " ".join(f"w{n}" for n in range(1, 26))
    seen = {word for chunk in chunk_words(text, 10, 2) for word in chunk.split()}
    assert seen == set(text.split())

def test_overlap_must_be_smaller_than_size():
    with pytest.raises(ValueError, match="smaller than size"):
        chunk_words("a b c", size=4, overlap=4)
''')

import sys
sys.path.insert(0, ".")
pytest.main(["-q", "test_chunking.py", "-p", "no:cacheprovider"])`} />

Five tests, no model, no network, a few milliseconds. `pytest.raises` asserts that a failure
happens *and* that it is the right one — a test that merely expects "an exception" passes when the
code raises `NameError` from a typo.

## Fixtures and parametrisation

A fixture builds what a test needs; `parametrize` runs one test over many cases:

<Run id="prod-test-3" code={`from pathlib import Path
import pytest, sys

Path("retrying.py").write_text('''
def should_retry(status):
    if status == 429:
        return True
    if 500 <= status < 600:
        return True
    if status in (408, 409):
        return True
    return False
''')

Path("test_retrying.py").write_text('''
import pytest
from retrying import should_retry

@pytest.fixture
def permanent_statuses():
    return [400, 401, 403, 404, 413, 422]

@pytest.mark.parametrize("status", [429, 500, 502, 503, 529, 408, 409])
def test_transient_statuses_are_retried(status):
    assert should_retry(status) is True

def test_permanent_statuses_are_not(permanent_statuses):
    for status in permanent_statuses:
        assert should_retry(status) is False, f"{status} should not be retried"

@pytest.mark.parametrize(("status", "expected"), [(429, True), (400, False), (503, True)])
def test_table(status, expected):
    assert should_retry(status) is expected
''')

sys.path.insert(0, ".")
pytest.main(["-q", "test_retrying.py", "-p", "no:cacheprovider"])`} />

Ten test cases from three functions. When one fails, pytest names the parameter that broke, so you
learn *which* status is mishandled rather than that "the test failed".

## Make the model injectable

Code that constructs its own client cannot be tested. Pass it in:

<Run id="prod-test-4" code={`from pathlib import Path
import pytest, sys

Path("summarise.py").write_text('''
REFUSAL = "I don\\'t know based on the provided documentation."

def summarise(client, text, *, model="claude-opus-5", max_words=20):
    """Note the client parameter: that is the seam a test needs."""
    if not text.strip():
        return ""
    response = client.create(
        model=model,
        max_tokens=256,
        system=f"Summarise in at most {max_words} words. Reply with the summary alone.",
        messages=[{"role": "user", "content": text}],
    )
    if response.stop_reason == "refusal":
        return REFUSAL
    return "".join(b.text for b in response.content if b.type == "text").strip()
''')

Path("test_summarise.py").write_text('''
import pytest
from types import SimpleNamespace
from summarise import summarise, REFUSAL

class FakeClient:
    """Records what it was asked; replies from a script."""
    def __init__(self, replies, stop_reason="end_turn"):
        self.replies = list(replies)
        self.stop_reason = stop_reason
        self.requests = []

    def create(self, **request):
        self.requests.append(request)
        return SimpleNamespace(
            content=[SimpleNamespace(type="text", text=self.replies.pop(0))],
            stop_reason=self.stop_reason,
        )

@pytest.fixture
def client():
    return FakeClient(["A short summary."])

def test_returns_the_text(client):
    assert summarise(client, "a long document") == "A short summary."

def test_sends_the_document_as_a_user_message(client):
    summarise(client, "a long document")
    request = client.requests[0]
    assert request["messages"][-1]["role"] == "user"
    assert request["messages"][-1]["content"] == "a long document"

def test_max_words_reaches_the_system_prompt(client):
    summarise(client, "text", max_words=7)
    assert "7 words" in client.requests[0]["system"]

def test_empty_input_never_calls_the_model(client):
    assert summarise(client, "   ") == ""
    assert client.requests == []

def test_refusal_is_passed_through():
    refusing = FakeClient(["irrelevant"], stop_reason="refusal")
    assert summarise(refusing, "text") == REFUSAL
''')

sys.path.insert(0, ".")
pytest.main(["-q", "test_summarise.py", "-p", "no:cacheprovider"])`} />

Look at what those tests actually check: that the document reaches the user message, that a
parameter reaches the system prompt, that empty input short-circuits, that a refusal is handled.
Every one is a real bug you can ship, and none of them needs a model.

<Callout type="ai" title="Why this matters for AI">
Nearly all the bugs in an LLM application are in the code around the model: the chunk that was
never indexed, the tool result appended to the wrong message, the retry that retried a 400, the
prompt where a placeholder rendered as a literal brace. All of those are deterministic, cheap to
test, and invisible to any evaluation of answer quality. Keep the seam — pass the client in — and
your suite stays fast enough to run on every save.
</Callout>

## Assert properties, not strings

For the output that genuinely is non-deterministic, assert what must always hold:

<Run id="prod-test-5" code={`from pathlib import Path
import pytest, sys

Path("answers.py").write_text('''
import re

REFUSAL = "I don\\'t know based on the provided documentation."

def check(answer, supplied_ids, *, max_words=60):
    cited = set(re.findall(r"\\\\[([a-zA-Z0-9_]+)\\\\]", answer))
    refused = answer.strip() == REFUSAL
    return {
        "refused": refused,
        "cited": cited,
        "unsupported": cited - set(supplied_ids),
        "words": len(answer.split()),
        "within_length": len(answer.split()) <= max_words,
    }
''')

Path("test_answers.py").write_text('''
import pytest
from answers import check, REFUSAL

# Recorded model outputs: what a real run produced, captured once.
SAMPLES = [
    ("Use an overlap of 10 to 20 percent [c1].", ["c1", "c2"]),
    ("Store vectors as float32 [c2], and normalise them [c2].", ["c1", "c2"]),
    (REFUSAL, ["c1"]),
]

@pytest.mark.parametrize(("answer", "supplied"), SAMPLES)
def test_never_cites_an_unsupplied_chunk(answer, supplied):
    assert check(answer, supplied)["unsupported"] == set()

@pytest.mark.parametrize(("answer", "supplied"), SAMPLES)
def test_answers_are_cited_or_refusals(answer, supplied):
    result = check(answer, supplied)
    assert result["refused"] or result["cited"], "an answer must cite something"

@pytest.mark.parametrize(("answer", "supplied"), SAMPLES)
def test_answers_stay_brief(answer, supplied):
    assert check(answer, supplied)["within_length"]

def test_a_fabricated_citation_is_caught():
    result = check("Store as float16 [c9].", ["c1", "c2"])
    assert result["unsupported"] == {"c9"}
''')

sys.path.insert(0, ".")
pytest.main(["-q", "test_answers.py", "-p", "no:cacheprovider"])`} />

"Cites only supplied chunks", "is a citation or a refusal", "is under sixty words" — those hold
for every correct answer, so they are testable. "Equals this exact sentence" is not.

<Callout type="warning">
Recorded samples go stale. They test your *checking* code, which is the point, but they say
nothing about whether today's model still behaves. That question belongs to the evaluation
harness in the next lesson, which runs against live calls on a schedule.
</Callout>

## Floats and markers

<Run id="prod-test-6" code={`from pathlib import Path
import pytest, sys

Path("test_misc.py").write_text('''
import pytest

def cosine(a, b):
    dot = sum(x * y for x, y in zip(a, b))
    left = sum(x * x for x in a) ** 0.5
    right = sum(y * y for y in b) ** 0.5
    return dot / (left * right)

def test_floats_need_a_tolerance():
    assert cosine([1.0, 0.0], [1.0, 0.0]) == pytest.approx(1.0)
    assert cosine([1.0, 1.0], [1.0, 0.0]) == pytest.approx(0.7071, abs=1e-4)

@pytest.mark.slow
def test_something_expensive():
    assert sum(range(1_000_00)) > 0

@pytest.mark.live
def test_calls_the_real_api():
    pytest.skip("needs ANTHROPIC_API_KEY and real money")
''')

Path("pytest.ini").write_text('''[pytest]
markers =
    slow: takes more than a second
    live: makes a real API call and costs money
''')

sys.path.insert(0, ".")
print("default run:")
pytest.main(["-q", "test_misc.py", "-p", "no:cacheprovider"])

print()
print("skipping the expensive ones:")
pytest.main(["-q", "test_misc.py", "-m", "not slow and not live", "-p", "no:cacheprovider"])`} />

Markers let one suite serve two purposes. In CI the default job runs
`pytest -m "not live"` on every pull request, and a nightly job runs the rest with a key.

<Callout type="tip">
Use `0.0 == pytest.approx(x)` for anything floating point. `0.1 + 0.2 == 0.3` is `False`, and a
cosine similarity computed two different ways will differ in the last few bits — a test that
demands exact equality fails for reasons that are not bugs.
</Callout>

## Recap

- Unit tests answer "is the code correct?"; evaluation answers "is the system good?". Keep them
  apart.
- Never call a live model in a unit test. Pass the client in as a parameter — that seam is what
  makes the code testable at all.
- `pytest.raises(Error, match=...)` checks the failure is the right one; `parametrize` turns one
  test into a table.
- For non-deterministic output, assert invariants: only supplied citations, length bounds, schema
  validity.
- `pytest.approx` for floats, and markers plus `-m "not live"` to keep the expensive tests out of
  the default run.
