<Callout type="tip" title="These apps really run">
FastAPI is available in this sandbox, but a web *server* is not — there are no sockets and no
threads here. The blocks below therefore call the ASGI application directly, which is exactly what
a server does after accepting a connection. The apps, the validation and the responses are real.
</Callout>

## The smallest useful service

<Run id="prod-api-1" code={`import json
from fastapi import FastAPI
from pydantic import BaseModel, Field

app = FastAPI(title="answers")

class AskRequest(BaseModel):
    question: str = Field(min_length=1, max_length=2000)
    max_words: int = Field(default=60, ge=10, le=500)

class AskResponse(BaseModel):
    answer: str
    citations: list[str] = Field(default_factory=list)
    model: str

@app.get("/healthz")
async def healthz():
    return {"status": "ok"}

@app.post("/ask", response_model=AskResponse)
async def ask(body: AskRequest) -> AskResponse:
    return AskResponse(
        answer=f"You asked: {body.question} (at most {body.max_words} words)",
        citations=["c1"],
        model="claude-opus-5",
    )

async def call(app, method, path, body=None):
    """Drive an ASGI app directly — what a server does once a request arrives."""
    payload = b"" if body is None else json.dumps(body).encode()
    scope = {
        "type": "http", "asgi": {"version": "3.0"}, "http_version": "1.1",
        "method": method, "path": path, "raw_path": path.encode(),
        "query_string": b"", "root_path": "", "scheme": "http",
        "headers": [(b"host", b"test"), (b"content-type", b"application/json"),
                    (b"content-length", str(len(payload)).encode())],
        "client": ("test", 1), "server": ("test", 80),
    }
    pending = [{"type": "http.request", "body": payload, "more_body": False}]
    async def receive():
        return pending.pop(0) if pending else {"type": "http.disconnect"}
    out = {"body": b"", "status": None}
    async def send(message):
        if message["type"] == "http.response.start":
            out["status"] = message["status"]
        elif message["type"] == "http.response.body":
            out["body"] += message.get("body", b"")
    await app(scope, receive, send)
    text = out["body"].decode()
    return out["status"], (json.loads(text) if text else None)

print(await call(app, "GET", "/healthz"))
print(await call(app, "POST", "/ask", {"question": "How much overlap?"}))`} />

Everything typed in `AskRequest` is enforced before your function runs. Everything in
`AskResponse` is enforced on the way out, so a handler cannot quietly return a shape the client
does not expect.

## Inject the client

The same seam as the testing lesson, expressed as a FastAPI dependency:

<Run id="prod-api-3" code={`import json
from types import SimpleNamespace
from fastapi import FastAPI, Depends
from pydantic import BaseModel

class FakeClient:
    def __init__(self, reply="A short answer [c1]."):
        self.reply = reply
        self.requests = []
    async def create(self, **request):
        self.requests.append(request)
        return SimpleNamespace(
            content=[SimpleNamespace(type="text", text=self.reply)],
            stop_reason="end_turn",
        )

async def get_client():
    """In production this returns anthropic.AsyncAnthropic(); tests override it.

    Declared async so FastAPI awaits it on the event loop. A synchronous
    dependency is run in a worker thread instead, which is wasted work when
    there is nothing blocking to do.
    """
    return FakeClient()

app = FastAPI()

class AskRequest(BaseModel):
    question: str

@app.post("/ask")
async def ask(body: AskRequest, client=Depends(get_client)):
    response = await client.create(
        model="claude-opus-5",

## Error handling

An upstream failure must become the right status code, not a 500:

<Run id="prod-api-4" code={`import json
from fastapi import FastAPI, HTTPException
from fastapi.responses import JSONResponse
from pydantic import BaseModel

class RateLimited(Exception):
    def __init__(self, retry_after=30):
        self.retry_after = retry_after
class Overloaded(Exception): pass
class InvalidRequest(Exception): pass

app = FastAPI()

@app.exception_handler(RateLimited)
async def rate_limited(request, error):
    return JSONResponse(status_code=429, content={"detail": "upstream rate limited"},
                        headers={"Retry-After": str(error.retry_after)})

@app.exception_handler(Overloaded)
async def overloaded(request, error):
    return JSONResponse(status_code=503, content={"detail": "upstream overloaded"})

@app.exception_handler(InvalidRequest)
async def invalid(request, error):
    return JSONResponse(status_code=400, content={"detail": str(error) or "invalid request"})

class AskRequest(BaseModel):
    question: str

@app.post("/ask")
async def ask(body: AskRequest):
    if body.question == "rate-limit":
        raise RateLimited(retry_after=12)
    if body.question == "overloaded":
        raise Overloaded()
    if body.question == "too-long":
        raise InvalidRequest("prompt exceeds the context window")
    if body.question == "not-found":
        raise HTTPException(status_code=404, detail="no such document")
    return {"answer": "fine"}

async def call(app, method, path, body=None):
    payload = b"" if body is None else json.dumps(body).encode()
    scope = {"type": "http", "asgi": {"version": "3.0"}, "http_version": "1.1",
             "method": method, "path": path, "raw_path": path.encode(),
             "query_string": b"", "root_path": "", "scheme": "http",
             "headers": [(b"host", b"t"), (b"content-type", b"application/json"),
                         (b"content-length", str(len(payload)).encode())],
             "client": ("t", 1), "server": ("t", 80)}
    pending = [{"type": "http.request", "body": payload, "more_body": False}]
    async def receive():
        return pending.pop(0) if pending else {"type": "http.disconnect"}
    out = {"body": b"", "status": None, "headers": []}
    async def send(message):
        if message["type"] == "http.response.start":
            out["status"] = message["status"]
            out["headers"] = message.get("headers", [])
        elif message["type"] == "http.response.body":
            out["body"] += message.get("body", b"")
    await app(scope, receive, send)
    text = out["body"].decode()
    return out, (json.loads(text) if text else None)

for question in ["fine", "rate-limit", "overloaded", "too-long", "not-found"]:
    out, payload = await call(app, "POST", "/ask", {"question": question})
    headers = {k.decode(): v.decode() for k, v in out["headers"]}
    extra = f"  Retry-After: {headers['retry-after']}" if "retry-after" in headers else ""
    print(f"{out['status']}  {question:<12} {payload}{extra}")`} />

<Callout type="warning">
Never let an upstream 429 surface as a 500. A 500 tells the caller "this is broken and retrying
will not help"; a 429 with `Retry-After` tells them exactly what to do. Mapping upstream failures
onto honest status codes is what makes your service usable by someone else's retry logic.
</Callout>

## Streaming

For a chat endpoint, streaming is the difference between a spinner and visible progress:

```python
import json
from fastapi import FastAPI
from fastapi.responses import StreamingResponse

app = FastAPI()

@app.post("/ask/stream")
async def ask_stream(body: AskRequest):
    async def events():
        async with client.messages.stream(
            model="claude-opus-5",
            max_tokens=4096,
            messages=[{"role": "user", "content": body.question}],
        ) as stream:
            async for text in stream.text_stream:
                yield f"data: {json.dumps({'delta': text})}\n\n"
            final = await stream.get_final_message()
            yield f"data: {json.dumps({'done': True, 'usage': final.usage.model_dump()})}\n\n"

    return StreamingResponse(events(), media_type="text/event-stream")
```

Server-sent events are the simplest transport that works: a plain HTTP response, one `data:` line
per chunk, and no WebSocket to manage. Send a final event carrying the usage so the client can
show what the answer cost.

## Health and readiness

Two checks, and they are not the same:

<Run id="prod-api-5" code={`import json
from fastapi import FastAPI
from fastapi.responses import JSONResponse

app = FastAPI()
state = {"index_loaded": False}

@app.get("/healthz")
async def healthz():
    """Is the process alive? Must never depend on anything external."""
    return {"status": "ok"}

@app.get("/readyz")
async def readyz():
    """Can it serve traffic? May check that dependencies are ready."""
    if not state["index_loaded"]:
        return JSONResponse(status_code=503, content={"status": "loading the index"})
    return {"status": "ready"}

async def call(app, method, path):
    scope = {"type": "http", "asgi": {"version": "3.0"}, "http_version": "1.1",
             "method": method, "path": path, "raw_path": path.encode(),
             "query_string": b"", "root_path": "", "scheme": "http",
             "headers": [(b"host", b"t")], "client": ("t", 1), "server": ("t", 80)}
    pending = [{"type": "http.request", "body": b"", "more_body": False}]
    async def receive():
        return pending.pop(0) if pending else {"type": "http.disconnect"}
    out = {"body": b"", "status": None}
    async def send(message):
        if message["type"] == "http.response.start":
            out["status"] = message["status"]
        elif message["type"] == "http.response.body":
            out["body"] += message.get("body", b"")
    await app(scope, receive, send)
    return out["status"], json.loads(out["body"].decode())

print("before the index loads:")
print(" ", await call(app, "GET", "/healthz"))
print(" ", await call(app, "GET", "/readyz"))

state["index_loaded"] = True
print("after:")
print(" ", await call(app, "GET", "/readyz"))`} />

A liveness check that calls the model API will restart your pods during an upstream incident. Keep
`/healthz` dependency-free and put the dependency checks in `/readyz`.

<Callout type="ai" title="Why this matters for AI">
An LLM endpoint has a latency profile nothing else in your stack shares: seconds rather than
milliseconds, and highly variable. That changes the operational defaults — timeouts measured in
tens of seconds, streaming so the connection never idles, a concurrency limit sized by your API
rate limit rather than by CPU, and a queue in front when demand exceeds it. Sizing a worker pool
the way you would for a database-backed endpoint gives you either idle capacity or a queue that
never drains.
</Callout>

Because the handlers are `async` and almost all of their time is spent waiting on the model API,
a small number of workers serves a large number of concurrent requests — the module's first lesson

## Recap

- Declare typed request and response models; validation, the 422 body and the OpenAPI schema all
  come from them.
- Inject the client with `Depends`, and swap it in tests with `dependency_overrides`.
- Map upstream failures onto honest status codes — a 429 with `Retry-After`, not a 500.
- Stream with `StreamingResponse` and server-sent events, and send a final event carrying usage.
- `/healthz` must be dependency-free; dependency checks belong in `/readyz`.
- LLM endpoints are seconds-long and variable: long timeouts, streaming, and concurrency sized by
  your rate limit rather than by CPU.
